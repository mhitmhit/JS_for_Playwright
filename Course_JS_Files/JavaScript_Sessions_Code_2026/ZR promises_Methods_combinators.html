Promise.race([espresso(), latte(), cappuccino()])
  .then(winner => console.log("Winner:", winner))
  .catch(err => console.log("First to fail:", err));

// After 1s (fastest one wins, even though others are still running):
Winner: "Cappuccino ready"

All possible scenarios
Scenario 1At least ONE resolves — rejections are ignoredAnimated above
Promise.any([espresso(), latte(), cappuccino()])
  .then(winner => console.log("First success:", winner))
  .catch(err => console.log("All failed:", err.errors));

// After 2s (Cappuccino's failure at 1s ignored, Espresso wins):
First success: "Espresso ready"
Scenario 2If ALL reject — rejects with AggregateError
Promise.any([espresso(), latte(), cappuccino()])
  .then(winner => console.log("First success:", winner))
  .catch(err => console.log("All failed:", err.errors));

// Waits for the slowest, then rejects with AggregateError:
All failed: [
  "Espresso machine broken",
  "No milk available",
  "Out of beans!"
]
